\subsection{Opción Múltiple — Una sola respuesta}

\begin{tcolorbox}[title=\textbf{Pregunta 1: Selección única}]
\textbf{¿Para qué sirve un certificado digital en una página web?}

\begin{enumerate}[label=\alph*)]
    \item Para que la página cargue más rápido
    \item Para permitir decoraciones visuales
    \item Para asegurar que la conexión sea privada y segura
    \item Para guardar archivos del usuario
    \item Para bloquear anuncios
\end{enumerate}

\textbf{Respuesta correcta:} \textcolor{blue}{c)}
\end{tcolorbox}

\subsection{Opción Múltiple — Varias respuestas correctas}

\begin{tcolorbox}[title=\textbf{Pregunta 2: Selecciones múltiples}]
\textbf{¿Qué beneficios obtiene un usuario cuando navega en un sitio web con conexión segura (HTTPS)?}

\begin{enumerate}[label=\alph*)]
    \item Su información viaja cifrada
    \item Puede verificar que está hablando con el sitio correcto
    \item Evita que terceros espíen sus datos
    \item Tiene internet más rápido automáticamente
    \item Las páginas cambian de diseño
\end{enumerate}

\textbf{Respuestas correctas:} \textcolor{blue}{a), b), c)}
\end{tcolorbox}
